cmake_minimum_required(VERSION 3.10)
project(zkGPT_shared)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_FLAGS "-mcmodel=large -O3 -lpthread -pthread -fPIC")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Include directories
include_directories(src)
include_directories(3rd)
include_directories(3rd/mcl/include)
link_directories(3rd/mcl)

# Build mcl library first
add_subdirectory(3rd/mcl)

# Collect all source files except main files
aux_source_directory(src gpt_src)
list(FILTER gpt_src EXCLUDE REGEX "main.*\.cpp$")

# Create shared library
add_library(zkGPT_shared SHARED ${gpt_src})
target_link_libraries(zkGPT_shared mcl)

# Set library properties
set_target_properties(zkGPT_shared PROPERTIES
    VERSION 1.0.0
    SOVERSION 1
    OUTPUT_NAME "zkgpt"
)

# Create a simple C wrapper for easier integration
add_library(zkGPT_wrapper SHARED src/wrapper.cpp)
target_link_libraries(zkGPT_wrapper zkGPT_shared mcl)

# Install targets
install(TARGETS zkGPT_shared zkGPT_wrapper
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

install(FILES src/wrapper.h
    DESTINATION include
)
